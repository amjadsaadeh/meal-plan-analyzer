```python
import pytest
from playwright.sync_api import Page, expect


@pytest.mark.django_db(transaction=True)
def test_export_button_visible(logged_in_page, live_server, meal_plan_with_day):
    """Export button renders on the meal plan detail page after Vue mounts."""
    plan, _ = meal_plan_with_day
    page = logged_in_page
    page.goto(f"{live_server.url}/meal-plan/{plan.pk}/")
    # Wait for Vue to mount — the button is rendered by ExportButton.vue
    export_btn = page.locator("button.btn-pdf")
    expect(export_btn).to_be_visible(timeout=5000)


@pytest.mark.django_db(transaction=True)
def test_export_button_error_state(logged_in_page, live_server, meal_plan_with_day):
    """Clicking export transitions to progress bar, then to error state on failed job."""
    plan, _ = meal_plan_with_day
    page = logged_in_page
    fake_job_id = "12345678-0000-0000-0000-000000000001"

    # Mock POST /api/export-jobs/ — only intercept POST, let GETs fall through
    def handle_post_or_continue(route):
        if route.request.method == "POST":
            route.fulfill(
                status=201,
                content_type="application/json",
                body=(
                    f'{{"id": "{fake_job_id}", "status": "pending",'
                    f' "progress": 0, "error_message": ""}}'
                ),
            )
        else:
            route.continue_()

    page.route("**/api/export-jobs/", handle_post_or_continue)

    # Mock GET poll — always return failed
    page.route(
        f"**/api/export-jobs/{fake_job_id}/",
        lambda route: route.fulfill(
            status=200,
            content_type="application/json",
            body=(
                f'{{"id": "{fake_job_id}", "status": "failed",'
                f' "progress": 0, "error_message": "Test failure message"}}'
            ),
        ),
    )

    page.goto(f"{live_server.url}/meal-plan/{plan.pk}/")
    page.wait_for_selector("button.btn-pdf", timeout=5000)
    page.locator("button.btn-pdf").click()

    # Progress bar should appear while pending, then error state after poll responds
    expect(page.locator(".export-error")).to_be_visible(timeout=5000)
    expect(page.locator(".export-error-msg")).to_contain_text("Test failure message")

    # Retry button must be visible in the error state
    retry_btn = page.locator(".export-error button.btn-pdf")
    expect(retry_btn).to_be_visible()
```

Key implementation decisions:
- Uses `transaction=True` on `django_db` — required by all frontend tests in this project (see existing tests pattern)
- `page.route()` intercepts by URL pattern before the request is sent to Django — no backend involvement for the export jobs API in these tests
- The POST handler uses `if route.request.method == "POST"` to distinguish from other requests to the same base URL (important: without this, GET requests to the endpoint would also be intercepted)
- `timeout=5000` gives Vue 5 seconds to mount and the interval timer 5 seconds to fire — should be more than sufficient for CI
- Only two tests: visibility and error flow. A "done" flow test would require mocking `window.location.href` assignment in Playwright, which is complex and not worth the added complexity here.
